// PlatoRelay GIF Captcha Solver (JavaScript port of deltax) // Supports driftodd and coherence captcha types 'use strict'; // ============ GIF Parsing (LZW decoder) ============ function parseGIF(buffer) { const data = new Uint8Array(buffer); i < clearCode; i++) dict.push([i]); dict.push([clearCode]); dict.push([eoiCode]); codeSize = minCodeSize + 1; prev = null; continue; } let output; if (code < dict.length) { output = dict[code]; } else if (code === dict.length && prev !== null) { output = prev.concat(prev[0]); } else { break; } for (let i = 0; i < output.length && pixIdx < pixelCount; i++) { pixels[pixIdx++] = output[i]; } if (prev !== null && dict.length < 4096) { dict.push(prev.concat(output[0])); if (dict.length === (1 << codeSize) && codeSize < 12) codeSize++; } prev = output; } return pixels; } // ============ Frame Composition ============ function compositeFrames(gif) { const { w, h, frames } = gif; const composed = []; let canvas = new Uint8ClampedArray(w * h * 4); (2 - maxC - minC) : (maxC + minC)); } function findBlobs(rgba, w, h, bgR, bgG, bgB, thresh = 22, minSize = 12, minSat = 0) { // Build foreground mask const fg = new Uint8Array(w * h); for (let y = 0; y < h; y++) { for (let x = 0; x < w; x++) { const i = (y * w + x) * 4; const diff = Math.abs(rgba[i] - bgR) + Math.abs(rgba[i + 1] - bgG) + Math.abs(rgba[i + 2] - bgB); fg[y * w + x] = diff > thresh ? 1 : 0; } } const visited = new Uint8Array(w * h); const blobs = []; for (let y = 0; y < h; y++) { for (let x = 0; x < w; x++) { const idx = y * w + x; if (!fg[idx] || visited[idx]) continue; // BFS const queue = [[x, y]]; const pixels = []; visited[idx] = 1; let ri = 0, gi = 0, bi = 0; while (queue.length) { const [cx, cy] = queue.shift(); pixels.push([cx, cy]); const pi = (cy * w + cx) * 4; ri += rgba[pi]; gi += rgba[pi + 1]; bi += rgba[pi + 2]; const neighbors = [[cx - 1, cy], [cx + 1, cy], [cx, cy - 1], [cx, cy + 1]]; for (const [nx, ny] of neighbors) { if (nx < 0 || nx >= w || ny < 0 || ny >= h) continue; const ni = ny * w + nx; if (!visited[ni] && fg[ni]) { visited[ni] = 1; queue.push([nx, ny]); } } } if (pixels.length < minSize) continue; const n = pixels.length; let cxMean = 0, cyMean = 0; for (const [px, py] of pixels) { cxMean += px; cyMean += py; } cxMean /= n; cyMean /= n; const mr = ri / n, mg = gi / n, mb = bi / n; const sat = saturation(mr, mg, mb); if (sat < minSat) continue; blobs.push({ cx: cxMean, cy: cyMean, size: n, r: mr, g: mg, b: mb, sat }); } } return blobs.sort((a, b) => b.size - a.size); } function detectSatThreshold(rgba, w, h, bgR, bgG, bgB) { const allBlobs = findBlobs(rgba, w, h, bgR, bgG, bgB, 22, 4, 0); if (allBlobs.length < 3) return 0; const sats = allBlobs.map(b => b.sat).sort((a, b) => a - b); let maxGap = 0, gapAt = 0; for (let i = 1; i < sats.length; i++) { const gap = sats[i] - sats[i - 1]; if (gap > maxGap) { maxGap = gap; gapAt = sats[i - 1]; } } const threshold = maxGap > 0.10 ?
